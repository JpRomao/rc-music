cmake_minimum_required(VERSION 3.31)

project(RCMusic LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CTest)
include(cmake/RcMusicCore.cmake)

option(RC_MUSIC_BUILD_BROWSER_INTEGRATION "Build the optional browser request boundary (no installed host)" OFF)
option(RC_MUSIC_BUILD_DOWNLOADS "Build optional download job coordination without Qt or network" OFF)
option(RC_MUSIC_BUILD_DOWNLOAD_NETWORK "Build optional direct-download transport (not connected to player)" OFF)
option(RC_MUSIC_BUILD_DOWNLOAD_PERSISTENCE "Build optional private download checkpoint adapters (not connected to player)" OFF)
option(RC_MUSIC_BUILD_NATIVE_MESSAGING "Build optional Windows Native Messaging host and dedicated download IPC" OFF)
option(RC_MUSIC_BUILD_EMBEDDED_BROWSER "Build experimental Windows WebView2 navigation (no downloads)" OFF)
set(RC_MUSIC_PLAYER_DOWNLOAD_PROFILE_HEADER "" CACHE FILEPATH
    "Explicit reviewed C++ player download profile header; empty keeps the normal player disabled")
if(RC_MUSIC_BUILD_DOWNLOADS OR RC_MUSIC_BUILD_BROWSER_INTEGRATION OR RC_MUSIC_BUILD_DOWNLOAD_NETWORK OR RC_MUSIC_BUILD_DOWNLOAD_PERSISTENCE OR RC_MUSIC_BUILD_NATIVE_MESSAGING)
    include(cmake/RcMusicDownloads.cmake)
endif()
if(RC_MUSIC_BUILD_DOWNLOAD_PERSISTENCE OR RC_MUSIC_BUILD_DOWNLOAD_NETWORK)
    include(cmake/RcMusicDownloadDocuments.cmake)
endif()
if(RC_MUSIC_BUILD_DOWNLOAD_PERSISTENCE)
    include(cmake/RcMusicDownloadPersistence.cmake)
endif()
if(RC_MUSIC_BUILD_BROWSER_INTEGRATION OR RC_MUSIC_BUILD_DOWNLOAD_NETWORK OR RC_MUSIC_BUILD_NATIVE_MESSAGING)
    include(cmake/RcMusicBrowserIntegration.cmake)
endif()
if(RC_MUSIC_BUILD_DOWNLOAD_NETWORK)
    include(cmake/RcMusicDownloadNetwork.cmake)
endif()
if(RC_MUSIC_BUILD_NATIVE_MESSAGING)
    include(cmake/RcMusicNativeMessaging.cmake)
endif()

option(RC_MUSIC_BUILD_DESKTOP "Build the Windows RC Music desktop application" OFF)
if(RC_MUSIC_BUILD_DESKTOP)
    if(NOT WIN32)
        message(FATAL_ERROR "The product desktop build currently targets Windows; Linux validation is deferred")
    endif()
    set(RC_MUSIC_DESKTOP_SHELL_SUPPORT ON)
    set(RC_MUSIC_DESKTOP_AUDIO_SUPPORT ON)
    add_subdirectory(src)
endif()
if(RC_MUSIC_BUILD_EMBEDDED_BROWSER)
    include(cmake/RcMusicEmbeddedBrowser.cmake)
endif()
if(RC_MUSIC_BUILD_DESKTOP AND WIN32 AND TARGET rc_music_download_network AND TARGET rc_music_download_persistence)
    include(cmake/RcMusicPlayerDownloads.cmake)
    rc_music_add_player_downloads(rc_music_player_downloads m0_qt_shell_ui)
endif()
if(NOT RC_MUSIC_PLAYER_DOWNLOAD_PROFILE_HEADER STREQUAL "")
    include(cmake/RcMusicPlayerDownloads.cmake)
    rc_music_configure_player_downloads(m0_qt_shell "${RC_MUSIC_PLAYER_DOWNLOAD_PROFILE_HEADER}")
endif()

if(BUILD_TESTING)
    set(RC_MUSIC_CATCH2_SOURCE_DIR "" CACHE PATH "Path to the preflight-verified Catch2 3.15.3 source tree")
    include(cmake/RcMusicTests.cmake)
    rc_music_prepare_catch2("${RC_MUSIC_CATCH2_SOURCE_DIR}")
    rc_music_add_core_tests(rc_music_core_tests "rc_music.core." "core")
    if(TARGET rc_music_downloads)
        add_executable(rc_music_downloads_tests
            tests/core/DownloadServiceTests.cpp tests/core/DownloadCheckpointTests.cpp
            tests/core/DownloadStagedAdmissionTests.cpp tests/core/DownloadAdmissionControllerTests.cpp
            tests/core/SingleWorkerDownloadCheckpointIoTests.cpp)
        target_link_libraries(rc_music_downloads_tests PRIVATE rc_music_downloads Catch2::Catch2WithMain)
        rc_music_configure_cpp(rc_music_downloads_tests)
        catch_discover_tests(rc_music_downloads_tests TEST_PREFIX "rc_music.downloads."
            PROPERTIES LABELS "downloads" TIMEOUT 15)
    endif()
    if(RC_MUSIC_BUILD_DESKTOP)
        include(cmake/RcMusicDesktopTests.cmake)
        rc_music_add_desktop_tests()
        add_subdirectory(tests/ui)
        add_subdirectory(tests/process)
    endif()
endif()
